    message(FATAL_ERROR "Native Metal 4 requires macOS on Apple Silicon")
endif()

if(NOT CMAKE_OSX_ARCHITECTURES)

    message(FATAL_ERROR "Native Metal 4 supports only arm64; Intel and universal targets are excluded")

# The emulator's parent build disables exceptions globally; native validation
# and failure recovery use C++ exceptions within the Metal implementation.
